View Integration Code & API Usage 
<!-- 1. Include the custom element bundle -->
<script src="./widget/plumber-viewer.umd.js"></script>

<!-- 2. Mount the element in your body -->
<div id="widget-host"></div>

<!-- 3. Initialize & control via JavaScript -->
<script>
  const viewer = PlumberViewer.create('#widget-host', {
    data: pipelineGraphData,
    theme: 'dark',
    isolation: true,
    documentation: true
  });

  // API call to focus a node
  document.getElementById('focus').onclick = () => {
    viewer.focusNode('Animation');
  };
</script>
